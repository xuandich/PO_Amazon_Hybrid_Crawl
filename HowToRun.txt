# Cài tất cả từ requirements.txt
pip install -r requirements.txt

# Build image
docker build -t amazon-scraper .

# Cấp quyền thực thi cho script
chmod +x run.sh

# Lấy cookie lần đầu (cần có proxy)
./run.sh cookie

# Chạy scrape dữ liệu
./run.sh scrape

# Hoặc chạy với số luồng tùy chỉnh
./run.sh scrape 20

# Chạy cả cookie + scrape
./run.sh all
